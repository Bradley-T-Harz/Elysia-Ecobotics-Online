def cf(path):
    req=urllib.request.Request('https://api.cloudflare.com/client/v4/accounts/'+ACCOUNT+path,headers={'Authorization':'Bearer '+token})
    with urllib.request.urlopen(req,timeout=30) as response: return json.load(response)['result']

def worker(name):
    settings=cf('/workers/scripts/'+name+'/settings')
    deployments=cf('/workers/scripts/'+name+'/deployments')['deployments']
    bindings=settings.get('bindings',[])
    values={b['name']:b.get('text') for b in bindings if b['type']=='plain_text'}
    return {'mode':values.get('BILLING_MODE'),'origin':values.get('BILLING_PUBLIC_ORIGIN',values.get('SANDBOX_UI_ORIGIN',values.get('IDENTITY_ALLOWED_ORIGINS'))),
        'database':values.get('SUPABASE_URL'),'flags':{k:v for k,v in values.items() if k.endswith('_ENABLED') or k.endswith('_CONFIRMED')},
        'sandboxAccessRequired':values.get('BILLING_SANDBOX_ACCESS_REQUIRED'),
        'accessConfigured':bool(values.get('SANDBOX_ACCESS_TEAM_DOMAIN') and values.get('SANDBOX_ACCESS_AUDIENCE')),
        'secretBindingNames':sorted(b['name'] for b in bindings if b['type']=='secret_text'),
        'services':[{'binding':b['name'],'service':b.get('service')} for b in bindings if b['type']=='service'],
        'assetBindingPresent':any(b['type']=='assets' for b in bindings),
        'subdomain':cf('/workers/scripts/'+name+'/subdomain'),
        'deployments':[{'id':d['id'],'created_on':d['created_on'],'versions':d['versions']} for d in deployments[:1]]}

def probe(case):
    base,path,method,expected,extra=case
    req=urllib.request.Request(base+path,method=method,headers={'User-Agent':'Elysia-sandbox-protection-verification/1.0',**extra},data=b'{}' if method=='POST' else None)
    try:response=urllib.request.urlopen(req,timeout=30)
    except urllib.error.HTTPError as e:response=e
    body=response.read()
    status=response.status
    assert status==expected, (base,path,status,expected)
    assert b'<html' not in body.lower() and b'<script' not in body.lower()
    return {'url':base+path,'method':method,'forgedAssertion':bool(extra),'status':status,'expectedStatus':expected,'bodySha256':hashlib.sha256(body).hexdigest(),'bytes':len(body),'cacheControl':response.headers.get('Cache-Control'),'xRobotsTag':response.headers.get('X-Robots-Tag'),'applicationBodyExposed':False}
